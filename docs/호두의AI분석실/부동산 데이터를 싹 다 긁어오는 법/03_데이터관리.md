```python
"""SQLite 데이터베이스 관리 모듈"""
import sqlite3
import hashlib
import os
import re
from datetime import datetime
from typing import Optional, Tuple, Dict, Any


class DatabaseManager:
    """SQLite 데이터베이스 관리 클래스"""

    def __init__(self, db_path: str = './data/naver_real_estate.db'):
        self.db_path = db_path
        os.makedirs(os.path.dirname(db_path), exist_ok=True)
        self._init_db()

    def _get_connection(self) -> sqlite3.Connection:
        conn = sqlite3.connect(self.db_path)
        conn.row_factory = sqlite3.Row
        return conn

    def _init_db(self):
        conn = self._get_connection()
        cursor = conn.cursor()

        cursor.execute('''
            CREATE TABLE IF NOT EXISTS properties (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                property_key TEXT UNIQUE NOT NULL,
                district TEXT NOT NULL,
                dong TEXT,
                property_name TEXT NOT NULL,
                detail_info TEXT,
                first_seen_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                last_seen_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                is_active INTEGER DEFAULT 1
            )
        ''')

        cursor.execute('''
            CREATE TABLE IF NOT EXISTS price_history (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                property_id INTEGER NOT NULL,
                trade_type TEXT NOT NULL,
                price TEXT NOT NULL,
                price_numeric INTEGER,
                description TEXT,
                additional_info TEXT,
                crawled_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (property_id) REFERENCES properties(id)
            )
        ''')

        cursor.execute('''
            CREATE TABLE IF NOT EXISTS crawl_logs (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                started_at TIMESTAMP NOT NULL,
                completed_at TIMESTAMP,
                total_districts INTEGER,
                total_properties INTEGER,
                new_properties INTEGER,
                status TEXT,
                error_message TEXT
            )
        ''')

        cursor.execute('''
            CREATE TABLE IF NOT EXISTS district_crawl_logs (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                crawl_log_id INTEGER NOT NULL,
                district TEXT NOT NULL,
                coords TEXT,
                properties_count INTEGER,
                status TEXT,
                error_message TEXT,
                completed_at TIMESTAMP,
                FOREIGN KEY (crawl_log_id) REFERENCES crawl_logs(id)
            )
        ''')

        # 인덱스 생성
        cursor.execute('CREATE INDEX IF NOT EXISTS idx_properties_district ON properties(district)')
        cursor.execute('CREATE INDEX IF NOT EXISTS idx_properties_active ON properties(is_active)')
        cursor.execute('CREATE INDEX IF NOT EXISTS idx_property_key ON properties(property_key)')

        conn.commit()
        conn.close()

    @staticmethod
    def generate_property_key(district: str, property_name: str, detail_info: str) -> str:
        raw_key = f"{district}|{property_name}|{detail_info}"
        return hashlib.md5(raw_key.encode()).hexdigest()

    @staticmethod
    def parse_price_to_numeric(price_text: str) -> Optional[int]:
        if not price_text:
            return None
        try:
            price_text = price_text.replace(',', '').strip()
            if '억' in price_text:
                parts = price_text.split('억')
                eok = int(parts[0].strip()) * 10000
                rest = 0
                if len(parts) > 1 and parts[1].strip():
                    rest_match = re.search(r'(\d+)', parts[1])
                    if rest_match:
                        rest = int(rest_match.group(1))
                return eok + rest
            else:
                match = re.search(r'(\d+)', price_text)
                return int(match.group(1)) if match else None
        except:
            return None

    def upsert_property(self, data: Dict[str, Any]) -> Tuple[int, bool]:
        property_key = self.generate_property_key(
            data.get('district', ''), data.get('property_name', ''), data.get('detail_info', '')
        )
        conn = self._get_connection()
        cursor = conn.cursor()

        try:
            cursor.execute("SELECT id FROM properties WHERE property_key = ?", (property_key,))
            existing = cursor.fetchone()

            if existing:
                property_id = existing['id']
                cursor.execute("UPDATE properties SET last_seen_at = CURRENT_TIMESTAMP, is_active = 1 WHERE id = ?", (property_id,))
                is_new = False
            else:
                cursor.execute(
                    "INSERT INTO properties (property_key, district, dong, property_name, detail_info) VALUES (?, ?, ?, ?, ?)",
                    (property_key, data.get('district', ''), data.get('dong', ''), data.get('property_name', ''), data.get('detail_info', ''))
                )
                property_id = cursor.lastrowid
                is_new = True

            price_numeric = self.parse_price_to_numeric(data.get('price', ''))
            cursor.execute(
                "INSERT INTO price_history (property_id, trade_type, price, price_numeric, description, additional_info) VALUES (?, ?, ?, ?, ?, ?)",
                (property_id, data.get('trade_type', ''), data.get('price', ''), price_numeric, data.get('description', ''), data.get('additional_info', ''))
            )
            conn.commit()
            return property_id, is_new
        finally:
            conn.close()

    def start_crawl_log(self, total_districts: int) -> int:
        conn = self._get_connection()
        cursor = conn.cursor()
        cursor.execute("INSERT INTO crawl_logs (started_at, total_districts, status) VALUES (?, ?, 'running')", (datetime.now().isoformat(), total_districts))
        crawl_log_id = cursor.lastrowid
        conn.commit()
        conn.close()
        return crawl_log_id

    def complete_crawl_log(self, crawl_log_id: int, result: Dict[str, Any]):
        conn = self._get_connection()
        cursor = conn.cursor()
        cursor.execute(
            "UPDATE crawl_logs SET completed_at = ?, total_properties = ?, new_properties = ?, status = ? WHERE id = ?",
            (datetime.now().isoformat(), result.get('total_properties', 0), result.get('new_properties', 0), result.get('status', 'unknown'), crawl_log_id)
        )
        conn.commit()
        conn.close()

    def log_district_crawl(self, crawl_log_id: int, district: str, coords: str, count: int, status: str, error: str = None):
        conn = self._get_connection()
        cursor = conn.cursor()
        cursor.execute(
            "INSERT INTO district_crawl_logs (crawl_log_id, district, coords, properties_count, status, error_message, completed_at) VALUES (?, ?, ?, ?, ?, ?, ?)",
            (crawl_log_id, district, coords, count, status, error, datetime.now().isoformat())
        )
        conn.commit()
        conn.close()

    def mark_inactive_properties(self, district: str, current_keys: set):
        if not current_keys:
            return
        conn = self._get_connection()
        cursor = conn.cursor()
        cursor.execute("SELECT id, property_key FROM properties WHERE district = ? AND is_active = 1", (district,))
        for row in cursor.fetchall():
            if row['property_key'] not in current_keys:
                cursor.execute("UPDATE properties SET is_active = 0 WHERE id = ?", (row['id'],))
        conn.commit()
        conn.close()

    def get_statistics(self) -> Dict[str, Any]:
        conn = self._get_connection()
        cursor = conn.cursor()
        stats = {}
        cursor.execute("SELECT COUNT(*) as cnt FROM properties")
        stats['total_properties'] = cursor.fetchone()['cnt']
        cursor.execute("SELECT COUNT(*) as cnt FROM properties WHERE is_active = 1")
        stats['active_properties'] = cursor.fetchone()['cnt']
        cursor.execute("SELECT district, COUNT(*) as cnt FROM properties WHERE is_active = 1 GROUP BY district ORDER BY cnt DESC")
        stats['properties_by_district'] = {row['district']: row['cnt'] for row in cursor.fetchall()}
        cursor.execute("SELECT * FROM crawl_logs ORDER BY started_at DESC LIMIT 5")
        stats['recent_crawls'] = [dict(row) for row in cursor.fetchall()]
        conn.close()
        return stats
```

## 🖥️ batch/cli_crawler.py

```python
#!/usr/bin/env python3
"""CLI 배치 크롤러"""
import argparse
import sys
import os

sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

from batch.crawler_core import NaverRealEstateCrawler
from batch.db_manager import DatabaseManager
from batch.logger import setup_logger

try:
    from config import SEOUL_DISTRICTS
except ImportError:
    SEOUL_DISTRICTS = [
        {"name": "강남구", "coords": "37.5172,127.0473,14"},
        {"name": "서초구", "coords": "37.4837,127.0324,14"},
        # ... 다른 구 추가
    ]


def main():
    parser = argparse.ArgumentParser(description='네이버 부동산 자동 크롤러')
    parser.add_argument('--districts', nargs='*', help='크롤링할 구 이름')
    parser.add_argument('--all', action='store_true', help='서울 전체 크롤링')
    parser.add_argument('--trade-type', default='A1:B1:B2', help='거래타입')
    parser.add_argument('--property-type', default='APT:APTHGJ', help='매물타입')
    parser.add_argument('--headless', action='store_true', default=True)
    parser.add_argument('--no-headless', action='store_true')
    parser.add_argument('--delay', type=float, default=3.0)
    parser.add_argument('--db-path', default='./data/naver_real_estate.db')
    parser.add_argument('--dry-run', action='store_true', help='테스트 모드')
    parser.add_argument('-v', '--verbose', action='store_true')
    parser.add_argument('--stats', action='store_true', help='DB 통계만 출력')

    args = parser.parse_args()
    logger = setup_logger(verbose=args.verbose)
    db = DatabaseManager(args.db_path)

    if args.stats:
        stats = db.get_statistics()
        print(f"총 매물: {stats['total_properties']:,}개")
        print(f"활성 매물: {stats['active_properties']:,}개")
        for d, c in stats['properties_by_district'].items():
            print(f"  {d}: {c:,}개")
        return 0

    if args.all:
        districts = SEOUL_DISTRICTS
    elif args.districts:
        district_names = set(args.districts)
        districts = [d for d in SEOUL_DISTRICTS if d['name'] in district_names]
    else:
        districts = SEOUL_DISTRICTS

    if not districts:
        logger.error("크롤링할 구가 없습니다.")
        return 1

    try:
        crawler = NaverRealEstateCrawler(
            db_manager=db,
            headless=not args.no_headless,
            delay_between_districts=args.delay,
            logger=logger
        )
        result = crawler.crawl_districts(
            districts=districts,
            trade_type=args.trade_type,
            property_type=args.property_type,
            dry_run=args.dry_run
        )
        return 0 if result['status'] in ['success', 'partial'] else 1

    except KeyboardInterrupt:
        logger.info("크롤링이 중단되었습니다.")
        return 130
    except Exception as e:
        logger.error(f"오류: {e}")
        return 1


if __name__ == '__main__':
    sys.exit(main())
```

## 🔧 batch/crawler_core.py (핵심 로직)

```python
"""크롤러 핵심 로직"""
import time
import random
from typing import List, Dict, Optional, Any
from datetime import datetime
from playwright.sync_api import sync_playwright, Browser, Page
from .db_manager import DatabaseManager


class NaverRealEstateCrawler:
    def __init__(self, db_manager: DatabaseManager, headless: bool = True, delay_between_districts: float = 3.0, logger=None):
        self.db = db_manager
        self.headless = headless
        self.delay = delay_between_districts
        self.logger = logger
        self.scroll_delay = 0.3
        self.page_load_delay = 10
        self.max_scroll_retries = 5

    def _log(self, msg: str, level: str = 'info'):
        if self.logger:
            getattr(self.logger, level)(msg)
        else:
            print(f"[{level.upper()}] {msg}")

    def _build_url(self, coords: str, property_type: str, trade_type: str, max_price: int = 0) -> str:
        url = f"https://new.land.naver.com/houses?ms={coords}&a={property_type}&b={trade_type}&e=RETAIL"
        if max_price > 0:
            url += f"&g={max_price}"
        return url

    def _launch_browser(self, playwright) -> Browser:
        for name in ["webkit", "chromium"]:
            for attempt in range(2):
                try:
                    browser = (playwright.webkit if name == "webkit" else playwright.chromium).launch(
                        headless=self.headless, args=['--no-sandbox', '--disable-dev-shm-usage']
                    )
                    ctx = browser.new_context()
                    pg = ctx.new_page()
                    pg.close()
                    ctx.close()
                    self._log(f"{name} 브라우저 실행 성공")
                    return browser
                except Exception as e:
                    self._log(f"{name} 시도 {attempt+1} 실패: {e}", 'warning')
                    time.sleep(1)
        raise RuntimeError("모든 브라우저 실행 실패")

    def _scroll_and_load_items(self, page: Page) -> List:
        scroll_sel, item_sel = ".item_list", ".item_inner"
        try:
            page.wait_for_selector(scroll_sel, timeout=10000)
        except:
            return []
        
        prev_count, retry_count = 0, 0
        while retry_count < self.max_scroll_retries:
            curr_count = page.evaluate(f'document.querySelectorAll("{item_sel}").length')
            if curr_count == 0:
                break
            if curr_count == prev_count:
                retry_count += 1
                time.sleep(1)
            else:
                retry_count = 0
            prev_count = curr_count
            page.evaluate(f'document.querySelector("{scroll_sel}").scrollTop = document.querySelector("{scroll_sel}").scrollHeight')
            time.sleep(self.scroll_delay)
        
        return page.evaluate(f'Array.from(document.querySelectorAll("{item_sel}")).map(i => ({{text: i.innerText}}))')

    def _parse_item(self, item, district: str) -> Optional[Dict[str, Any]]:
        try:
            text = item.get('text', '') if isinstance(item, dict) else str(item)
            lines = [l.strip() for l in text.split('\n') if l.strip()]
            if len(lines) < 3:
                return None
            
            name = lines[0]
            for prefix in ["집주인", "단지", "중개", "공인"]:
                if name.startswith(prefix):
                    name = name[len(prefix):]
                    break
            
            price_text, trade_type, price = lines[1], "", lines[1]
            for t in ["매매", "전세", "월세", "단기"]:
                if price_text.startswith(t):
                    trade_type, price = t, price_text[len(t):]
                    break
            
            return {"district": district, "dong": "", "property_name": name.strip(), "trade_type": trade_type, 
                    "price": price.strip(), "detail_info": lines[2], "description": lines[3] if len(lines) > 3 else "",
                    "additional_info": " | ".join(lines[4:]) if len(lines) > 4 else ""}
        except:
            return None

    def crawl_single_district(self, district: Dict, property_type: str, trade_type: str, dry_run: bool = False) -> Dict:
        result = {"district": district['name'], "status": "failed", "count": 0, "new_count": 0, "error": None}
        url = self._build_url(district['coords'], property_type, trade_type)
        self._log(f"크롤링: {district['name']}")

        with sync_playwright() as p:
            try:
                browser = self._launch_browser(p)
            except Exception as e:
                result['error'] = str(e)
                return result

            try:
                ctx = browser.new_context(viewport={'width': 1920, 'height': 1080}, 
                    user_agent='Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36')
                page = ctx.new_page()
                page.add_init_script("Object.defineProperty(navigator, 'webdriver', {get: () => undefined});")
                page.goto(url, timeout=60000)
                time.sleep(self.page_load_delay)

                items = self._scroll_and_load_items(page)
                if not items:
                    result['status'] = 'success'
                    return result

                property_keys = set()
                for item in items:
                    parsed = self._parse_item(item, district['name'])
                    if parsed:
                        if not dry_run:
                            _, is_new = self.db.upsert_property(parsed)
                            key = self.db.generate_property_key(parsed['district'], parsed['property_name'], parsed['detail_info'])
                            property_keys.add(key)
                            if is_new:
                                result['new_count'] += 1
                        result['count'] += 1

                if not dry_run and property_keys:
                    self.db.mark_inactive_properties(district['name'], property_keys)

                result['status'] = 'success'
                self._log(f"{district['name']}: {result['count']}개 ({result['new_count']}개 신규)")

            except Exception as e:
                result['error'] = str(e)
            finally:
                browser.close()
        return result

    def crawl_districts(self, districts: List, trade_type: str, property_type: str, dry_run: bool = False) -> Dict:
        crawl_log_id = self.db.start_crawl_log(len(districts)) if not dry_run else None
        total = {"status": "success", "total_districts": len(districts), "completed_districts": 0, 
                 "total_properties": 0, "new_properties": 0, "failed_districts": []}

        self._log(f"===== {len(districts)}개 구 크롤링 시작 =====")

        for i, d in enumerate(districts):
            self._log(f"[{i+1}/{len(districts)}] {d['name']}")
            r = self.crawl_single_district(d, property_type, trade_type, dry_run)

            if r['status'] == 'success':
                total['completed_districts'] += 1
                total['total_properties'] += r['count']
                total['new_properties'] += r['new_count']
                if not dry_run:
                    self.db.log_district_crawl(crawl_log_id, d['name'], d['coords'], r['count'], 'success')
            else:
                total['failed_districts'].append(d['name'])
                if not dry_run:
                    self.db.log_district_crawl(crawl_log_id, d['name'], d['coords'], 0, 'failed', r['error'])

            if i < len(districts) - 1:
                delay = self.delay + random.uniform(0, 2)
                self._log(f"다음 구 대기: {delay:.1f}초")
                time.sleep(delay)

        if total['failed_districts']:
            total['status'] = 'partial'

        self._log(f"===== 완료: {total['completed_districts']}/{total['total_districts']}개 구, {total['total_properties']}개 매물 =====")

        if not dry_run:
            self.db.complete_crawl_log(crawl_log_id, total)
        return total
```
