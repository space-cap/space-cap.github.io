## 📝 batch/logger.py

```python
"""
로깅 설정 모듈
"""
import logging
import os
from datetime import datetime
from logging.handlers import RotatingFileHandler


def setup_logger(verbose: bool = False, log_dir: str = './logs') -> logging.Logger:
    """
    로거 설정

    Args:
        verbose: True면 DEBUG 레벨, False면 INFO 레벨
        log_dir: 로그 파일 저장 디렉토리

    Returns:
        설정된 로거 객체
    """
    os.makedirs(log_dir, exist_ok=True)

    logger = logging.getLogger('naver_crawler')
    logger.setLevel(logging.DEBUG if verbose else logging.INFO)

    # 기존 핸들러 제거 (중복 방지)
    if logger.handlers:
        logger.handlers.clear()

    # 포맷 설정
    formatter = logging.Formatter(
        '%(asctime)s [%(levelname)s] %(message)s',
        datefmt='%Y-%m-%d %H:%M:%S'
    )

    # 파일 핸들러 (일별 로그, 최대 10MB, 30개 보관)
    log_file = os.path.join(log_dir, f"crawler_{datetime.now().strftime('%Y%m%d')}.log")
    file_handler = RotatingFileHandler(
        log_file,
        maxBytes=10 * 1024 * 1024,  # 10MB
        backupCount=30,
        encoding='utf-8'
    )
    file_handler.setFormatter(formatter)
    file_handler.setLevel(logging.DEBUG)
    logger.addHandler(file_handler)

    # 콘솔 핸들러
    console_handler = logging.StreamHandler()
    console_handler.setFormatter(formatter)
    console_handler.setLevel(logging.DEBUG if verbose else logging.INFO)
    logger.addHandler(console_handler)

    return logger
```

## 🔧 install.sh

```bash
#!/bin/bash

echo "🏠 네이버 부동산 크롤러 설치 스크립트"
echo "=========================================="
echo ""

# Python 버전 확인
python_version=$(python3 --version 2>&1)
if [ $? -eq 0 ]; then
    echo "✅ Python 설치 확인: $python_version"
else
    echo "❌ Python3가 설치되어 있지 않습니다."
    echo "   Python을 먼저 설치해주세요: https://www.python.org/downloads/"
    exit 1
fi

echo ""
echo "📦 Python 패키지 설치 중..."
pip3 install -r requirements.txt

if [ $? -eq 0 ]; then
    echo "✅ Python 패키지 설치 완료"
else
    echo "❌ Python 패키지 설치 실패"
    exit 1
fi

echo ""
echo "🌐 Playwright 브라우저 설치 중..."
playwright install chromium

if [ $? -eq 0 ]; then
    echo "✅ Playwright 브라우저 설치 완료"
else
    echo "❌ Playwright 브라우저 설치 실패"
    exit 1
fi

echo ""
echo "=========================================="
echo "✨ 설치 완료!"
echo "=========================================="
echo ""
echo "웹 앱을 실행하려면:"
echo "  streamlit run web_app.py"
echo ""
echo "설정을 변경하려면:"
echo "  config.py 파일을 편집하세요"
echo ""
```

## 🚀 빠른 시작 가이드

### 1. 설치

```bash
# 설치 스크립트 실행
chmod +x install.sh
./install.sh

# 또는 직접 설치
pip3 install -r requirements.txt
python3 -m playwright install chromium
```

### 2. 웹 앱 실행

```bash
# 웹 앱 실행
streamlit run web_app.py

# 또는 스크립트로 실행
chmod +x 웹실행.sh
./웹실행.sh
```

### 3. CLI 배치 크롤링

```bash
# 서울 전체 크롤링
python -m batch.cli_crawler --all

# 특정 구만 크롤링
python -m batch.cli_crawler --districts 강남구 서초구

# 테스트 모드 (DB 저장 안함)
python -m batch.cli_crawler --districts 동대문구 --dry-run
```
