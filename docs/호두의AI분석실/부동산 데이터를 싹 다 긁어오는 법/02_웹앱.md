```python
"""
네이버 부동산 크롤러 - 웹 버전
Streamlit 기반
"""

import streamlit as st
import time
import os
from datetime import datetime
import pandas as pd
from playwright.sync_api import sync_playwright

import sys
sys.path.append(os.path.dirname(os.path.abspath(__file__)))
from batch.db_manager import DatabaseManager
from config import CRAWLER_CONFIG

# 페이지 설정
st.set_page_config(page_title="네이버 부동산 크롤러", page_icon="🏠", layout="wide")
st.title("🏠 네이버 부동산 크롤러")
st.markdown("---")

# 사이드바 - 설정
with st.sidebar:
    st.header("⚙️ 크롤링 설정")
    
    st.subheader("📍 지역")
    col1, col2 = st.columns(2)
    with col1:
        district = st.text_input("구", value="동대문구")
    with col2:
        dong = st.text_input("동", value="답십리동")
    
    coords = st.text_input("좌표 (위도,경도,줌)", value="37.5665,127.0587,15")
    st.markdown("---")
    
    st.subheader("🏘️ 매물 타입")
    property_apt = st.checkbox("아파트", value=True)
    property_villa = st.checkbox("빌라/연립/다세대", value=False)
    property_house = st.checkbox("단독/다가구", value=False)
    property_shop = st.checkbox("상가주택", value=False)
    st.markdown("---")
    
    st.subheader("💰 거래 타입")
    trade_sell = st.checkbox("매매", value=True)
    trade_jeon = st.checkbox("전세", value=True)
    trade_month = st.checkbox("월세", value=False)
    st.markdown("---")
    
    st.subheader("💵 가격 설정")
    max_price = st.number_input("최대 가격 (만원)", min_value=0, value=150000, step=10000)
    st.caption(f"💡 {max_price/10000:.1f}억원")
    st.markdown("---")
    
    st.subheader("🔍 필터 키워드")
    keywords_input = st.text_input("키워드 (쉼표로 구분)", value="")
    st.caption("비우면 모든 매물 수집")
    st.markdown("---")

    st.subheader("🖥️ 브라우저 설정")
    headless_mode = st.checkbox("헤드리스 모드", value=False)

# 메인 영역
col_left, col_right = st.columns([2, 1])

with col_left:
    st.header("📊 크롤링 실행")
    
    with st.expander("🔧 현재 설정 보기", expanded=False):
        location = f"{district} {dong}".strip()
        st.write(f"**지역:** {location}")
        st.write(f"**좌표:** {coords}")
        
        property_types = []
        if property_apt: property_types.append("아파트")
        if property_villa: property_types.append("빌라/연립/다세대")
        if property_house: property_types.append("단독/다가구")
        if property_shop: property_types.append("상가주택")
        st.write(f"**매물 타입:** {', '.join(property_types) if property_types else '없음'}")
        
        trade_types = []
        if trade_sell: trade_types.append("매매")
        if trade_jeon: trade_types.append("전세")
        if trade_month: trade_types.append("월세")
        st.write(f"**거래 타입:** {', '.join(trade_types) if trade_types else '없음'}")
        st.write(f"**최대 가격:** {max_price:,}만원 ({max_price/10000:.1f}억원)")
    
    if st.button("🚀 크롤링 시작", type="primary", use_container_width=True):
        if not coords:
            st.error("❌ 좌표를 입력해주세요.")
            st.stop()
        if not property_types:
            st.error("❌ 최소 하나의 매물 타입을 선택해주세요.")
            st.stop()
        if not trade_types:
            st.error("❌ 최소 하나의 거래 타입을 선택해주세요.")
            st.stop()
        
        # 코드 생성
        property_codes = []
        if property_apt: property_codes.append("APT:APTHGJ")
        if property_villa: property_codes.append("VL:JWJT")
        if property_house: property_codes.append("DDDGG")
        if property_shop: property_codes.append("SGJT")
        property_code = ":".join(property_codes)
        
        trade_codes = []
        if trade_sell: trade_codes.append("A1")
        if trade_jeon: trade_codes.append("B1")
        if trade_month: trade_codes.append("B2")
        trade_code = ":".join(trade_codes)
        
        keywords = [k.strip() for k in keywords_input.split(",") if k.strip()] if keywords_input else []
        
        location = f"{district}_{dong}" if dong else district
        url = f"https://new.land.naver.com/houses?ms={coords}&a={property_code}&b={trade_code}&e=RETAIL&g={max_price}"
        
        progress_bar = st.progress(0)
        status_text = st.empty()
        
        try:
            status_text.info("🌐 브라우저 실행 중...")
            progress_bar.progress(10)
            results = []
            
            with sync_playwright() as p:
                browser = None
                browser_launched = False
                
                for browser_name in ["webkit", "chromium"]:
                    for attempt in range(2):
                        try:
                            if browser_name == "webkit":
                                browser = p.webkit.launch(headless=headless_mode, args=['--no-sandbox'])
                            else:
                                browser = p.chromium.launch(headless=headless_mode, args=['--no-sandbox', '--disable-gpu'])
                            
                            test_context = browser.new_context()
                            test_page = test_context.new_page()
                            test_page.close()
                            test_context.close()
                            browser_launched = True
                            st.success(f"✅ {browser_name} 브라우저 실행 성공!")
                            break
                        except Exception as e:
                            st.warning(f"⚠️ {browser_name} 시도 실패: {str(e)[:50]}")
                            time.sleep(1)
                    if browser_launched:
                        break

                if not browser_launched:
                    st.error("❌ 브라우저 실행 실패")
                    st.stop()
                    
                context = browser.new_context(
                    viewport={'width': 1920, 'height': 1080},
                    user_agent='Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'
                )
                page = context.new_page()
                page.add_init_script("Object.defineProperty(navigator, 'webdriver', {get: () => undefined});")
                
                status_text.info(f"📄 페이지 로딩 중...")
                progress_bar.progress(20)
                page.goto(url, timeout=60000)
                
                status_text.info("⏳ 페이지 안정화 대기 중...")
                for i in range(10):
                    progress_bar.progress(20 + i * 2)
                    time.sleep(1)
                
                # 동일매물 묶기 활성화
                try:
                    result = page.evaluate("""() => {
                        const cb = document.querySelector('#address_group2');
                        if (cb && !cb.checked) {
                            cb.checked = true;
                            cb.dispatchEvent(new Event('change', { bubbles: true }));
                            return 'activated';
                        }
                        return cb ? 'already_checked' : 'not_found';
                    }""")
                    if result == 'activated':
                        time.sleep(3)
                except:
                    pass
                
                progress_bar.progress(40)
                status_text.info("📜 스크롤하여 매물 로딩 중...")
                
                scroll_selector = ".item_list"
                item_selector = ".item_inner"
                
                try:
                    page.wait_for_selector(scroll_selector, timeout=10000)
                except:
                    st.error("❌ 리스트를 찾을 수 없습니다")
                    browser.close()
                    st.stop()
                
                prev_count, retry_count = 0, 0
                while retry_count < 5:
                    items = page.query_selector_all(item_selector)
                    curr_count = len(items)
                    
                    if curr_count == 0:
                        break
                    if curr_count == prev_count:
                        retry_count += 1
                        time.sleep(1)
                    else:
                        retry_count = 0
                    
                    prev_count = curr_count
                    page.evaluate(f'document.querySelector("{scroll_selector}").scrollTop = document.querySelector("{scroll_selector}").scrollHeight')
                    time.sleep(0.3)
                
                st.write(f"✅ {prev_count}개 매물 로딩됨")
                progress_bar.progress(60)
                
                items = page.query_selector_all(item_selector)
                if not items:
                    browser.close()
                    status_text.warning("⚠️ 매물이 없습니다.")
                    st.stop()
                
                status_text.success(f"✅ {len(items)}개 매물 발견!")
                progress_bar.progress(70)
                
                count = 0
                for idx, item in enumerate(items, 1):
                    try:
                        text = item.inner_text()
                        lines = [l.strip() for l in text.split('\n') if l.strip()]
                        if len(lines) < 3:
                            continue
                        
                        if keywords and not any(k in " ".join(lines) for k in keywords):
                            continue
                        
                        count += 1
                        district_gu, district_dong = (location.split("_", 1) + [""])[:2]
                        
                        property_name = lines[0]
                        for prefix in ["집주인", "단지", "중개", "공인"]:
                            if property_name.startswith(prefix):
                                property_name = property_name[len(prefix):]
                                break
                        
                        price_text = lines[1]
                        trade_type, price_value = "", price_text
                        for t in ["매매", "전세", "월세", "단기"]:
                            if price_text.startswith(t):
                                trade_type, price_value = t, price_text[len(t):]
                                break
                        
                        results.append({
                            "번호": count, "구": district_gu, "동": district_dong,
                            "매물명": property_name.strip(), "거래타입": trade_type,
                            "가격": price_value.strip(), "상세정보": lines[2],
                            "설명": lines[3] if len(lines) > 3 else "",
                            "추가정보": " | ".join(lines[4:]),
                            "수집일시": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                        })
                    except:
                        continue
                
                browser.close()
                progress_bar.progress(95)
                
                if results:
                    os.makedirs("./results", exist_ok=True)
                    filename = f"{location}_{datetime.now().strftime('%Y%m%d_%H%M%S')}.xlsx"
                    filepath = os.path.join("./results", filename)
                    
                    df = pd.DataFrame(results)
                    df.to_excel(filepath, index=False, engine='openpyxl')
                    
                    progress_bar.progress(100)
                    status_text.success(f"🎉 크롤링 완료! {count}개 매물 수집")
                    st.balloons()
                    
                    st.markdown("---")
                    st.subheader(f"📊 수집 결과 ({count}개)")
                    st.dataframe(df, use_container_width=True, height=400)
                    
                    with open(filepath, 'rb') as f:
                        st.download_button("📥 엑셀 파일 다운로드", f, filename,
                            "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
                            use_container_width=True)
                    st.success(f"💾 파일 저장: {filepath}")
                else:
                    status_text.warning("⚠️ 조건에 맞는 매물이 없습니다.")
        
        except Exception as e:
            status_text.error(f"❌ 오류 발생: {str(e)}")
            st.exception(e)

with col_right:
    st.header("📚 사용 가이드")
    
    with st.expander("🗺️ 좌표 찾는 방법"):
        st.markdown("""
        1. 네이버 지도에서 원하는 지역 검색
        2. URL에서 좌표 확인
        3. 형식: `위도,경도,줌레벨`
        
        **예시:** 답십리 `37.5665,127.0587,15`
        """)
    
    with st.expander("💡 사용 팁"):
        st.markdown("""
        **가격:** 전월세 30000~50000, 매매 50000~150000
        **키워드:** 신축, 풀옵션, 역세권 등
        """)

st.markdown("---")
st.caption("💻 네이버 부동산 크롤러 v1.0")
```
